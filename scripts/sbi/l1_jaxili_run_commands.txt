  --gpu 2 --no-wandb --map-kind nbody \
  --tfds-name NbodyCosmogridDatasetTomo/grid --field-size 10 --field-npix 80 \
  --nbins 4 --tomo-bin-indices 1,2,3,4 \
  --n-scales 5 --l1-nbins 40 --l1-min-snr -13 --l1-max-snr 13 \
  --summary-transform log1p-zscore --clip-value 5 --pca-components 50 \
  --epochs 500 --batch-size 256 --learning-rate 1e-4 \
  --npe-warmup-steps 256 --npe-decay-steps 200000 \
  --npe-samples 100000 --seed 42 \
  --cache-dir scripts/sbi/l1_jax_runs/cache_tomo4_10deg80_snr13 \
  --save-dir scripts/sbi/l1_jax_runs/tomo4_10deg80_seed42 \
  --checkpoint-name l1_jax_snr13_n5_b40 \
  --posterior-out scripts/sbi/l1_jax_runs/posteriors/l1_jax_tomo4_10deg80_seed42.npy \
  --plot --figure-out scripts/sbi/l1_jax_runs/figures/l1_jax_tomo4_10deg80_seed42.png
